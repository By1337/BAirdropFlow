                                        int chance = item.getAsInt("chance");
                                        int minAmount = item.getAsInt("minAmount");
                                        int maxAmount = item.getAsInt("maxAmount");
                                        ItemStack itemStack = BLib.getApi().getParseCompoundTag().create((CompoundTag) item.getOrThrow("item"));
                                        res.add(new LootEntry(new ItemHolder(itemStack), chance, minAmount, maxAmount));
                                    }
                                    int rolls = 25;

                                    if (cfg.exists()){
                                        rolls = YamlMap.load(cfg).get("genItemCount").asInt(25);
                                    }
                                    LootTable lootTable = new LootTable(res, rolls);
                                    System.out.println("adapt " + air.getName());
                                    LootTables.DEFAULT.set(air.getName(), lootTable);
                                }
                                LootTables.DEFAULT.save();

                                throw new RuntimeException(e);
                            }
